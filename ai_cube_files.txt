    # ========================================================
    # ROS IMAGE
    # ========================================================

    def ros_bgr(
        self,
        msg,
    ):

        if msg.encoding != "bgr8":

            raise RuntimeError(
                f"Expected bgr8, "
                f"got {msg.encoding}"
            )

        raw = np.frombuffer(
            msg.data,
            dtype=np.uint8,
        )

        expected = (
            msg.height
            *
            msg.step
        )

        if raw.size < expected:

            raise RuntimeError(
                "Za malo danych obrazu."
            )

        rows = raw[
            :expected
        ].reshape(
            msg.height,
            msg.step,
        )

        row_bytes = (
            msg.width
            *
            3
        )

        image = rows[
            :,
            :row_bytes
        ].reshape(
            msg.height,
            msg.width,
            3,
        )

        return np.ascontiguousarray(
            image
        )


    def ros_xyz(
        self,
        msg,
    ):

        if msg.encoding != "32FC3":

            raise RuntimeError(
                f"Expected 32FC3, "
                f"got {msg.encoding}"
            )

        raw = np.frombuffer(
            msg.data,
            dtype=np.float32,
        )

        expected = (
            msg.height
            *
            msg.width
            *
            3
        )

        if raw.size < expected:

            raise RuntimeError(
                "Za malo danych XYZ."
            )

        xyz = raw[
            :expected
        ].reshape(
            msg.height,
            msg.width,
            3,
        )

        return np.ascontiguousarray(
            xyz
        )


    # ========================================================
    # LETTERBOX
    # ========================================================

    def letterbox(
        self,
        image,
    ):

        original_h, original_w = (
            image.shape[:2]
        )

        scale = min(
            MODEL_SIZE / original_w,
            MODEL_SIZE / original_h,
        )

        new_w = int(
            round(
                original_w
                *
                scale
            )
        )

        new_h = int(
            round(
                original_h
                *
                scale
            )
        )

        resized = cv2.resize(
            image,
            (
                new_w,
                new_h,
            ),
            interpolation=
                cv2.INTER_LINEAR,
        )

        pad_w = (
            MODEL_SIZE
            -
            new_w
        )

        pad_h = (
            MODEL_SIZE
            -
            new_h
        )

        left = (
            pad_w // 2
        )

        right = (
            pad_w
            -
            left
        )

        top = (
            pad_h // 2
        )

        bottom = (
            pad_h
            -
            top
        )

        padded = cv2.copyMakeBorder(
            resized,
            top,
            bottom,
            left,
            right,
            cv2.BORDER_CONSTANT,
            value=(
                114,
                114,
                114,
            ),
        )

        return (
            padded,
            scale,
            left,
            top,
        )


    # ========================================================
    # PREPROCESS
    # ========================================================

    def preprocess(
        self,
        image,
    ):

        padded, scale, pad_x, pad_y = (
            self.letterbox(
                image
            )
        )

        rgb = cv2.cvtColor(
            padded,
            cv2.COLOR_BGR2RGB,
        )

        tensor = (
            rgb
            .astype(
                np.float32
            )
            /
            255.0
        )

        tensor = np.transpose(
            tensor,
            (
                2,
                0,
                1,
            ),
        )

        tensor = np.expand_dims(
            tensor,
            axis=0,
        )

        tensor = np.ascontiguousarray(
            tensor,
            dtype=np.float32,
        )

        return (
            tensor,
            scale,
            pad_x,
            pad_y,
        )


    # ========================================================
    # NMS
    # ========================================================

    def nms(
        self,
        boxes,
        scores,
    ):

        if len(
            boxes
        ) == 0:

            return []

        boxes_xywh = []

        for box in boxes:

            x1, y1, x2, y2 = box

            boxes_xywh.append(
                [
                    float(x1),
                    float(y1),
                    float(
                        x2 - x1
                    ),
                    float(
                        y2 - y1
                    ),
                ]
            )

        indices = cv2.dnn.NMSBoxes(
            boxes_xywh,
            scores,
            CONF_THRESHOLD,
            NMS_THRESHOLD,
        )

        if len(
            indices
        ) == 0:

            return []

        return [
            int(i)
            for i in np.array(
                indices
            ).reshape(-1)
        ]


    # ========================================================
    # SIGMOID
    # ========================================================

    @staticmethod
    def sigmoid(
        value,
    ):

        value = np.clip(
            value,
            -50.0,
            50.0,
        )

        return (
            1.0
            /
            (
                1.0
                +
                np.exp(
                    -value
                )
            )
        )


    # ========================================================
    # MASK
    # ========================================================

    def build_mask(
        self,
        coefficients,
        prototypes,
        box_model,
        original_shape,
        scale,
        pad_x,
        pad_y,
    ):

        # prototypes:
        # [32, 160, 160]

        channels, proto_h, proto_w = (
            prototypes.shape
        )

        proto_flat = (
            prototypes.reshape(
                channels,
                -1,
            )
        )

        mask = (
            coefficients
            @
            proto_flat
        )

        mask = self.sigmoid(
            mask
        )

        mask = mask.reshape(
            proto_h,
            proto_w,
        )

        # Prototype -> 640 x 640
        mask_640 = cv2.resize(
            mask,
            (
                MODEL_SIZE,
                MODEL_SIZE,
            ),
            interpolation=
                cv2.INTER_LINEAR,
        )

        x1, y1, x2, y2 = (
            box_model
        )

        x1 = int(
            max(
                0,
                min(
                    MODEL_SIZE - 1,
                    round(x1),
                ),
            )
        )

        y1 = int(
            max(
                0,
                min(
                    MODEL_SIZE - 1,
                    round(y1),
                ),
            )
        )

        x2 = int(
            max(
                x1 + 1,
                min(
                    MODEL_SIZE,
                    round(x2),
                ),
            )
        )

        y2 = int(
            max(
                y1 + 1,
                min(
                    MODEL_SIZE,
                    round(y2),
                ),
            )
        )

        crop = np.zeros(
            (
                MODEL_SIZE,
                MODEL_SIZE,
            ),
            dtype=np.float32,
        )

        crop[
            y1:y2,
            x1:x2
        ] = (
            mask_640[
                y1:y2,
                x1:x2
            ]
        )

        original_h, original_w = (
            original_shape
        )

        scaled_w = int(
            round(
                original_w
                *
                scale
            )
        )

        scaled_h = int(
            round(
                original_h
                *
                scale
            )
        )

        x_start = int(
            pad_x
        )

        y_start = int(
            pad_y
        )

        x_end = (
            x_start
            +
            scaled_w
        )

        y_end = (
            y_start
            +
            scaled_h
        )

        crop = crop[
            y_start:y_end,
            x_start:x_end
        ]

        if crop.size == 0:

            return None

        mask_original = cv2.resize(
            crop,
            (
                original_w,
                original_h,
            ),
            interpolation=
                cv2.INTER_LINEAR,
        )

        return (
            mask_original
            >
            MASK_THRESHOLD
        )


    # ========================================================
    # POSTPROCESS
    # ========================================================

    def postprocess(
        self,
        output0,
        output1,
        image_shape,
        scale,
        pad_x,
        pad_y,
    ):

        predictions = (
            output0[0]
            .transpose(
                1,
                0,
            )
        )

        prototypes = (
            output1[0]
        )

        boxes_model = []
        boxes_original = []
        scores = []
        coefficients = []

        original_h, original_w = (
            image_shape
        )

        for prediction in predictions:

            confidence = float(
                prediction[4]
            )

            if (
                confidence
                <
                CONF_THRESHOLD
            ):

                continue

            cx = float(
                prediction[0]
            )

            cy = float(
                prediction[1]
            )

            width = float(
                prediction[2]
            )

            height = float(
                prediction[3]
            )

            x1_model = (
                cx
                -
                width / 2.0
            )

            y1_model = (
                cy
                -
                height / 2.0
            )

            x2_model = (
                cx
                +
                width / 2.0
            )

            y2_model = (
                cy
                +
                height / 2.0
            )

            x1 = (
                x1_model
                -
                pad_x
            ) / scale

            y1 = (
                y1_model
                -
                pad_y
            ) / scale

            x2 = (
                x2_model
                -
                pad_x
            ) / scale

            y2 = (
                y2_model
                -
                pad_y
            ) / scale

            x1 = float(
                np.clip(
                    x1,
                    0,
                    original_w - 1,
                )
            )

            y1 = float(
                np.clip(
                    y1,
                    0,
                    original_h - 1,
                )
            )

            x2 = float(
                np.clip(
                    x2,
                    0,
                    original_w - 1,
                )
            )

            y2 = float(
                np.clip(
                    y2,
                    0,
                    original_h - 1,
                )
            )

            if (
                x2 <= x1
                or
                y2 <= y1
            ):

                continue

            boxes_model.append(
                [
                    x1_model,
                    y1_model,
                    x2_model,
                    y2_model,
                ]
            )

            boxes_original.append(
                [
                    x1,
                    y1,
                    x2,
                    y2,
                ]
            )

            scores.append(
                confidence
            )

            coefficients.append(
                prediction[
                    5:37
                ].astype(
                    np.float32
                )
            )

        keep = self.nms(
            boxes_original,
            scores,
        )

        detections = []

        for index in keep:

            mask = self.build_mask(
                coefficients[index],
                prototypes,
                boxes_model[index],
                image_shape,
                scale,
                pad_x,
                pad_y,
            )

            if mask is None:

                continue

            if (
                np.count_nonzero(
                    mask
                )
                <
                20
            ):

                continue

            detections.append(
                {
                    "box":
                        boxes_original[index],

                    "score":
                        float(
                            scores[index]
                        ),

                    "mask":
                        mask,
                }
            )

        detections.sort(
            key=lambda item:
                item["score"],
            reverse=True,
        )

        return detections


    # ========================================================
    # MASK -> XYZ
    # ========================================================

    def mask_to_xyz(
        self,
        mask,
        xyz_image,
    ):

        points = (
            xyz_image[
                mask
            ]
        )

        if (
            points.shape[0]
            <
            MIN_VALID_3D_POINTS
        ):

            return None

        valid = np.all(
            np.isfinite(
                points
            ),
            axis=1,
        )

        valid &= (
            points[:, 2]
            >
            MIN_DEPTH_M
        )

        valid &= (
            points[:, 2]
            <
            MAX_DEPTH_M
        )

        points = (
            points[
                valid
            ]
        )

        if (
            points.shape[0]
            <
            MIN_VALID_3D_POINTS
        ):

            return None

        # Najpierw odrzucamy punkty o skrajnej głębokości.
        z = points[
            :,
            2
        ]

        z_low = np.percentile(
            z,
            DEPTH_PERCENTILE_LOW,
        )

        z_high = np.percentile(
            z,
            DEPTH_PERCENTILE_HIGH,
        )

        points = points[
            (
                points[:, 2]
                >=
                z_low
            )
            &
            (
                points[:, 2]
                <=
                z_high
            )
        ]

        if (
            points.shape[0]
            <
            MIN_VALID_3D_POINTS
        ):

            return None

        center = np.median(
            points,
            axis=0,
        )

        return (
            center.astype(
                np.float64
            ),
            int(
                points.shape[0]
            ),
        )


    # ========================================================
    # TF CAMERA -> BASE
    # ========================================================

    def transform_to_base(
        self,
        point_camera,
    ):

        try:

            transform = (
                self.tf_buffer
                .lookup_transform(
                    BASE_FRAME,
                    CAMERA_FRAME,
                    rclpy.time.Time(),
                )
            )

        except Exception as exc:

            self.get_logger().warn(
                f"Brak TF "
                f"{CAMERA_FRAME}->{BASE_FRAME}: "
                f"{exc}"
            )

            return None

        t = (
            transform
            .transform
            .translation
        )

        q = (
            transform
            .transform
            .rotation
        )

        x = q.x
        y = q.y
        z = q.z
        w = q.w

        rotation = np.array(
            [
                [
                    1 - 2 * (y*y + z*z),
                    2 * (x*y - z*w),
                    2 * (x*z + y*w),
                ],
                [
                    2 * (x*y + z*w),
                    1 - 2 * (x*x + z*z),
                    2 * (y*z - x*w),
                ],
                [
                    2 * (x*z - y*w),
                    2 * (y*z + x*w),
                    1 - 2 * (x*x + y*y),
                ],
            ],
            dtype=np.float64,
        )

        translation = np.array(
            [
                t.x,
                t.y,
                t.z,
            ],
            dtype=np.float64,
        )

        camera = np.array(
            point_camera,
            dtype=np.float64,
        )

        return (
            rotation
            @
            camera
            +
            translation
        )


    # ========================================================
    # DEBUG
    # ========================================================

    def publish_debug(
        self,
        image,
        source_msg,
    ):

        success, encoded = cv2.imencode(
            ".jpg",
            image,
            [
                int(
                    cv2.IMWRITE_JPEG_QUALITY
                ),
                85,
            ],
        )

        if not success:

            return

        msg = (
            CompressedImage()
        )

        msg.header = (
            source_msg.header
        )

        msg.header.frame_id = (
            CAMERA_FRAME
        )

        msg.format = "jpeg"

        msg.data = (
            encoded.tobytes()
        )

        self.debug_pub.publish(
            msg
        )


    # ========================================================
    # NO DETECTION
    # ========================================================

    def publish_not_detected(

        detected = Bool()
        detected.data = False

        self.detected_pub.publish(
            detected
        )

        confidence = Float32()
        confidence.data = 0.0

        self.confidence_pub.publish(
            confidence
        )

        cv2.putText(
            debug,
            "AI CUBE: ---",
            (
                10,
                25,
            ),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.65,
            (
                0,
                0,
                255,
            ),
            2,
            cv2.LINE_AA,
        )

        self.publish_debug(
            debug,
            image_msg,
        )


    # ========================================================
    # CALLBACK
    # ========================================================

    def callback(
        self,
        image_msg,
        points_msg,
    ):

        try:

            image = self.ros_bgr(
                image_msg
            )

            xyz_image = self.ros_xyz(
                points_msg
            )

        except Exception as exc:

            self.get_logger().error(
                f"Input error: {exc}"
            )

            return

        if (
            image.shape[:2]
            !=
            xyz_image.shape[:2]
        ):

            self.get_logger().error(
                "Obraz i XYZ maja inne wymiary."
            )

            return

        debug = (
            image.copy()
        )

        # ====================================================
        # INFERENCE
        # ====================================================

        tensor, scale, pad_x, pad_y = (
            self.preprocess(
                image
            )
        )

        try:

            output0, output1 = (
                self.session.run(
                    [
                        self.output_detection_name,
                        self.output_proto_name,
                    ],
                    {
                        self.input_name:
                            tensor
                    },
                )
            )

        except Exception as exc:

            self.get_logger().error(
                f"ONNX inference error: "
                f"{exc}"
            )

            return

        detections = self.postprocess(
            output0,
            output1,
            image.shape[:2],
            scale,
            pad_x,
            pad_y,
        )

        if not detections:

            self.publish_not_detected(
                debug,
                image_msg,
            )

            return

        # Najpewniejsza kostka.
        detection = (
            detections[0]
        )

        score = float(
            detection[
                "score"
            ]
        )

        mask = (
            detection[
                "mask"
            ]
        )

        box = (
            detection[
                "box"
            ]
        )

        result_3d = self.mask_to_xyz(
            mask,
            xyz_image,
        )

        if result_3d is None:

            self.get_logger().warn(
                "AI widzi kostke, "
                "ale brak stabilnej glebi."
            )

            self.publish_not_detected(
                debug,
                image_msg,
            )

            return

        (
            camera_center,
            valid_points,
        ) = result_3d

        # ====================================================
        # CAMERA POINT
        # ====================================================

        camera_msg = (
            PointStamped()
        )

        camera_msg.header = (
            image_msg.header
        )

        camera_msg.header.frame_id = (

    # ========================================================
    # ROS IMAGE -> BGR
    # ========================================================

    def ros_image_to_bgr(
        self,
        msg,
    ):

        height = int(
            msg.height
        )

        width = int(
            msg.width
        )

        step = int(
            msg.step
        )


        if msg.encoding not in (
            "bgr8",
            "rgb8",
        ):

            raise RuntimeError(
                f"Nieobslugiwane encoding: "
                f"{msg.encoding}"
            )


        expected_row_bytes = (
            width
            *
            3
        )


        raw = np.frombuffer(
            msg.data,
            dtype=np.uint8,
        )


        expected_total = (
            step
            *
            height
        )


        if raw.size < expected_total:

            raise RuntimeError(
                "Za malo danych obrazu."
            )


        rows = (
            raw[
                :expected_total
            ]
            .reshape(
                height,
                step,
            )
        )


        pixels = rows[
            :,
            :expected_row_bytes
        ]


        image = pixels.reshape(
            height,
            width,
            3,
        )


        if msg.encoding == "rgb8":

            image = (
                cv2.cvtColor(
                    image,
                    cv2.COLOR_RGB2BGR,
                )
            )


        return np.ascontiguousarray(
            image
        )

    # ========================================================
    # CENTER DISTANCE
    # ========================================================

    def center_distance(
        self,
        points_3d,
        disparity,
    ):

        height, width = (
            disparity.shape
        )

        cx = (
            width // 2
        )

        cy = (
            height // 2
        )

        radius = 10

        x1 = max(
            0,
            cx - radius,
        )

        x2 = min(
            width,
            cx + radius + 1,
        )

        y1 = max(
            0,
            cy - radius,
        )

        y2 = min(
            height,
            cy + radius + 1,
        )


        disparity_roi = disparity[
            y1:y2,
            x1:x2,
        ]


        points_roi = points_3d[
            y1:y2,
            x1:x2,
            :
        ]


        z = points_roi[
            :,
            :,
            2
        ]


        valid = (
            np.isfinite(
                z
            )
            &
            (
                z > 0.05
            )
            &
            (
                z < 10.0
            )
            &
            (
                disparity_roi > 1.0
            )
        )


        values = z[
            valid
        ]


        if values.size < 10:

            return None


        return float(
            np.median(
                values
            )
        )


    # ========================================================
    # CALLBACK
    # ========================================================

    def stereo_callback(
        self,
        left_msg,
        right_msg,
    ):

        if not self.maps_ready:
            return


        try:

            left = (
                self.ros_image_to_bgr(
                    left_msg
                )
            )

            right = (
                self.ros_image_to_bgr(
                    right_msg
                )
            )

        except Exception as exc:

            self.get_logger().error(
                f"Konwersja obrazu: {exc}"
            )

            return


        if (
            left.shape[1],
            left.shape[0],
        ) != self.image_size:

            self.get_logger().error(
                "Rozdzielczosc lewej kamery "
                "nie pasuje do kalibracji."
            )

            return


        if (
            right.shape[1],
            right.shape[0],
        ) != self.image_size:

            self.get_logger().error(
                "Rozdzielczosc prawej kamery "
                "nie pasuje do kalibracji."
            )

            return


        # ====================================================
        # RAW DISPLAY PAIR
        # ====================================================

        left_display = (
            cv2.flip(
                left,
                DISPLAY_FLIP_CODE,
            )
        )

        right_display = (
            cv2.flip(
                right,
                DISPLAY_FLIP_CODE,
            )
        )

        cameras = np.hstack(
            (
                left_display,
                right_display,
            )
        )

        self.publish_compressed(
            self.cameras_pub,
            cameras,
            right_msg,
        )


        # ====================================================
        # RECTIFICATION
        # ====================================================

        left_rect = (
            cv2.remap(

                left,

                self.left_map_x,

                self.left_map_y,

                cv2.INTER_LINEAR,
            )
        )


        right_rect = (
            cv2.remap(

                right,

                self.right_map_x,

                self.right_map_y,

                cv2.INTER_LINEAR,
            )
        )


        # Machine-readable rectified left.
        self.publish_bgr_image(
            self.left_rect_pub,
            left_rect,
            left_msg,
        )


        # ====================================================
        # RECTIFIED DEBUG
        # ====================================================

        left_debug = (
            cv2.flip(
                left_rect,
                DISPLAY_FLIP_CODE,
            )
        )

        right_debug = (
            cv2.flip(
                right_rect,
                DISPLAY_FLIP_CODE,
            )
        )

        debug = np.hstack(
            (
                left_debug,
                right_debug,
            )
        )


        debug_h, debug_w = (
            debug.shape[:2]
        )


        for y in range(
            20,
            debug_h,
            20,
        ):

            cv2.line(
                debug,
                (
                    0,
                    y,
                ),
                (
                    debug_w - 1,
                    y,
                ),
                (
                    0,
                    255,
                    0,
                ),
                1,
            )


        cv2.line(
            debug,
            (
                left_debug.shape[1],
                0,
            ),
            (
                left_debug.shape[1],
                debug_h - 1,
            ),
            (
                0,
                0,
                255,
            ),
            1,
        )


        self.publish_compressed(
            self.rectified_debug_pub,
            debug,
            right_msg,
        )


        # ====================================================
        # GRAY
        # ====================================================

        left_gray = (
            cv2.cvtColor(
                left_rect,
                cv2.COLOR_BGR2GRAY,
            )
        )

        right_gray = (
            cv2.cvtColor(
                right_rect,
                cv2.COLOR_BGR2GRAY,
            )
        )


        # ====================================================
        # DISPARITY
        # ====================================================

        disparity_raw = (
            self.stereo.compute(
                left_gray,
                right_gray,
            )
        )


        disparity = (
            disparity_raw.astype(
                np.float32
            )
            /
            16.0
        )


        # ====================================================
        # XYZ
        # ====================================================

        points_3d = (
            cv2.reprojectImageTo3D(

                disparity,

                self.q_matrix,

                handleMissingValues=True,
            )
        )


        points_3d = (
            points_3d.astype(
                np.float32
            )
        )


        # ====================================================
        # INVALID -> NaN
        # ====================================================

        invalid = (
            ~np.isfinite(
                points_3d[
                    :,
                    :,
                    2
                ]
            )
            |
            (
                disparity <= 1.0
            )
            |
            (
                points_3d[
                    :,
                    :,
                    2

import cv2
import numpy as np
import rclpy

from rclpy.node import Node
from sensor_msgs.msg import CompressedImage


class DualCameraNode(Node):
    def __init__(self):
        super().__init__("dual_camera_node")

        self.left_publisher = self.create_publisher(
            CompressedImage,
            "/camera_left/image_raw/compressed",
            5,
        )

        self.right_publisher = self.create_publisher(
            CompressedImage,
            "/camera_right/image_raw/compressed",
            5,
        )

        # Najczęściej kamera 0 i kamera 1.
        # Jeśli będą zamienione, zamień liczby miejscami.
        self.left_camera = cv2.VideoCapture(0)
        self.right_camera = cv2.VideoCapture(1)

        self.configure_camera(self.left_camera)
        self.configure_camera(self.right_camera)

        if not self.left_camera.isOpened():
            self.get_logger().error(
                "Nie można otworzyć lewej kamery /dev/video0"
            )

        if not self.right_camera.isOpened():
            self.get_logger().error(
                "Nie można otworzyć prawej kamery /dev/video1"
            )

        # Około 10 klatek na sekundę.
        self.timer = self.create_timer(
            0.1,
            self.timer_callback,
        )

        self.get_logger().info(
            "Dual Camera Node uruchomiony"
        )

    def configure_camera(self, camera):
        camera.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
        camera.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)
        camera.set(cv2.CAP_PROP_FPS, 10)

    def create_compressed_message(self, frame):
        message = CompressedImage()

        message.header.stamp = (
            self.get_clock()
            .now()
            .to_msg()
        )

        message.format = "jpeg"

        success, compressed = cv2.imencode(
            ".jpg",
            frame,
            [
                int(cv2.IMWRITE_JPEG_QUALITY),
                70,
            ],
        )

        if not success:
            return None

        message.data = np.asarray(
            compressed,
            dtype=np.uint8,
        ).tobytes()

        return message

    def timer_callback(self):
        left_success, left_frame = self.left_camera.read()
        right_success, right_frame = self.right_camera.read()

        if left_success:
            left_message = self.create_compressed_message(
                left_frame
            )

            if left_message is not None:
                self.left_publisher.publish(
                    left_message
                )

        if right_success:
            right_message = self.create_compressed_message(
                right_frame
            )

            if right_message is not None:
                self.right_publisher.publish(
                    right_message
                )

    def destroy_node(self):
        self.left_camera.release()
        self.right_camera.release()

        super().destroy_node()


def main(args=None):
    rclpy.init(args=args)

    node = DualCameraNode()

    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()

        if rclpy.ok():
            rclpy.shutdown()


if __name__ == "__main__":
    main()

    def stamp_to_seconds(
        self,
        stamp,
    ):

        return (
            float(stamp.sec)
            +
            float(stamp.nanosec) / 1e9
        )


    # ========================================================
    # UJEDNOLICENIE KOLEJNOSCI NAROZNIKOW
    # ========================================================


    def ros_image_to_bgr(
        self,
        msg,
    ):
        if msg.encoding != "bgr8":
            raise RuntimeError(
                f"Nieobslugiwane encoding: {msg.encoding}"
            )

        height = int(msg.height)
        width = int(msg.width)
        step = int(msg.step)

        expected_row_bytes = width * 3

        if step < expected_row_bytes:
            raise RuntimeError(
                f"Niepoprawny step={step}, "
                f"oczekiwano co najmniej "
                f"{expected_row_bytes}"
            )

        raw = np.frombuffer(
            msg.data,
            dtype=np.uint8,
        )

        expected_total = (
            step
            *
            height
        )

        if raw.size < expected_total:
            raise RuntimeError(
                f"Za malo danych: {raw.size}, "
                f"oczekiwano {expected_total}"
            )

        rows = raw[
            :expected_total
        ].reshape(
            height,
            step,
        )

        pixels = rows[
            :,
            :expected_row_bytes
        ]

        image = pixels.reshape(
            height,
            width,
            3,
        )

        return np.ascontiguousarray(
            image
        )



    def canonicalize_corners(
        self,
        corners,
    ):

        grid = corners.reshape(
            BOARD_ROWS,
            BOARD_COLS,
            2
        ).copy()


        first_row_y = np.mean(
            grid[0, :, 1]
        )

        last_row_y = np.mean(
            grid[-1, :, 1]
        )


        if first_row_y > last_row_y:

            grid = np.flip(
                grid,
                axis=0
            )


        first_col_x = np.mean(
            grid[:, 0, 0]
        )

        last_col_x = np.mean(
            grid[:, -1, 0]
        )


        if first_col_x > last_col_x:

            grid = np.flip(
                grid,
                axis=1
            )


        return grid.reshape(
            -1,
            1,
            2
        ).astype(
            np.float32
        )


    # ========================================================
    # DETEKCJA SZACHOWNICY
    # ========================================================

    def find_board(
        self,
        image,
    ):

        gray = cv2.cvtColor(
            image,
            cv2.COLOR_BGR2GRAY
        )


        flags = (
            cv2.CALIB_CB_NORMALIZE_IMAGE
            |
            cv2.CALIB_CB_EXHAUSTIVE
            |
            cv2.CALIB_CB_ACCURACY
        )


        found, corners = (
            cv2.findChessboardCornersSB(
                gray,
                (
                    BOARD_COLS,
                    BOARD_ROWS,
                ),
                flags=flags,
            )
        )


        if not found:

            return None


        if corners is None:

            return None


        if len(corners) != (
            BOARD_COLS * BOARD_ROWS
        ):

            return None


        return self.canonicalize_corners(
            corners
        )


    # ========================================================
    # OPIS POLOZENIA PLANSZY
    # ========================================================

    def make_descriptor(
        self,
        corners,
        width,
        height,
    ):

        points = corners.reshape(
            -1,
            2
        )


        indices = [
            0,
            BOARD_COLS - 1,
            (
                (BOARD_ROWS - 1)
                *
                BOARD_COLS
            ),
            (
                BOARD_ROWS
                *
                BOARD_COLS
                -
                1
            ),
        ]


        descriptor = []


        for index in indices:

            descriptor.append(
                float(
                    points[index][0]
                    /
                    width
                )
            )

            descriptor.append(
                float(
                    points[index][1]
                    /
                    height
                )
            )


        return np.array(
            descriptor,
            dtype=np.float32
        )


    # ========================================================
    # POLE PLANSZY
    # ========================================================

    def board_area_ratio(
        self,
        corners,
        width,
        height,
    ):

        points = corners.reshape(
            -1,
            2
        )


        indices = [
            0,
            BOARD_COLS - 1,
            (
                BOARD_ROWS
                *
                BOARD_COLS
                -
                1
            ),
            (
                (BOARD_ROWS - 1)
                *
                BOARD_COLS
            ),
        ]


        polygon = points[
            indices
        ].astype(
            np.float32
        )


        area = abs(
            cv2.contourArea(
                polygon
            )
        )


        image_area = float(
            width * height
        )


        return area / image_area


    # ========================================================
    # CZY POLOZENIE JEST NOWE
    # ========================================================

    def is_diverse(
        self,
        descriptor,
    ):

        if not self.accepted_descriptors:

            return True


        distances = [
            float(
                np.linalg.norm(
                    descriptor
                    -
                    old
                )
            )
            for old
            in self.accepted_descriptors
        ]


        return (
            min(distances)
            >
            DIVERSITY_THRESHOLD
        )


    # ========================================================
    # STABILNOSC PLANSZY
    # ========================================================

    def update_stability(
        self,
        descriptor,
    ):

        if self.last_candidate_descriptor is None:

            self.last_candidate_descriptor = (
                descriptor.copy()
            )

            self.stable_counter = 1

            return False


        difference = float(
            np.linalg.norm(
                descriptor
                -
                self.last_candidate_descriptor
            )
        )


        if difference < STABILITY_THRESHOLD:

            self.stable_counter += 1

        else:

            self.stable_counter = 1


        self.last_candidate_descriptor = (
            descriptor.copy()
        )


        return (
            self.stable_counter
            >=
            STABLE_FRAMES_REQUIRED
        )


    def reset_stability(
        self,
    ):

        self.last_candidate_descriptor = None
        self.stable_counter = 0

    # ========================================================
    # CALLBACK
    # ========================================================

    def stereo_callback(
        self,
        left_msg,
        right_msg,
    ):

        if self.done:

            return


        if self.calibrating:

            return


        now = time.monotonic()


        if (
            now
            -
            self.last_accept_time
            <
            SAMPLE_COOLDOWN_SEC
        ):

            return


        # ====================================================
        # KONWERSJA OBRAZOW
        # ====================================================

        try:

            left = self.ros_image_to_bgr(
                left_msg
            )

            right = self.ros_image_to_bgr(
                right_msg
            )

        except Exception as exc:

            self.get_logger().error(
                f"cv_bridge: {exc}"
            )

            return


        if (
            left.shape[:2]
            !=
            right.shape[:2]
        ):

            self.get_logger().error(
                "Rozne rozdzielczosci kamer."
            )

            return


        height, width = (
            left.shape[:2]
        )


        if self.image_size is None:

            self.image_size = (
                width,
                height
            )

            self.get_logger().info(
                f"Rozdzielczosc: "
                f"{width}x{height}"
            )


        # ====================================================
        # CZAS
        # ====================================================

        left_time = (
            self.stamp_to_seconds(
                left_msg.header.stamp
            )
        )

        right_time = (
            self.stamp_to_seconds(
                right_msg.header.stamp
            )
        )


        time_difference = abs(
            left_time
            -
            right_time
        )


        # Dodatkowe reczne zabezpieczenie.
        if time_difference > SYNC_SLOP_SEC:

            self.reset_stability()

            if (
                now
                -
                self.last_status_time
                >
                1.5
            ):

                self.get_logger().info(
                    f"Odrzucam pare: "
                    f"dt={time_difference * 1000:.1f} ms"
                )

                self.last_status_time = now

            return


        # ====================================================
        # NAROZNIKI
        # ====================================================

        left_corners = self.find_board(
            left
        )

        right_corners = self.find_board(
            right
        )


        if (
            left_corners is None
            or
            right_corners is None
        ):

            self.reset_stability()


            if (
                now
                -
                self.last_status_time
                >
                2.0
            ):

                self.get_logger().info(
                    f"Szukam planszy... "
                    f"{len(self.object_points)}/"
                    f"{TARGET_SAMPLES}"
                )

                self.last_status_time = now


            return


        # ====================================================
        # WIELKOSC PLANSZY
        # ====================================================

        left_area = self.board_area_ratio(
            left_corners,
            width,
            height,
        )

        right_area = self.board_area_ratio(
            right_corners,
            width,
            height,
        )


        if (
            left_area < MIN_BOARD_AREA_RATIO
            or
            right_area < MIN_BOARD_AREA_RATIO
        ):

            self.reset_stability()


            if (
                now
                -
                self.last_status_time
                >
                1.5
            ):

                self.get_logger().info(
                    f"Plansza za mala: "
                    f"L={left_area * 100:.1f}% "
                    f"R={right_area * 100:.1f}%. "
                    f"Podejdz blizej."
                )

                self.last_status_time = now


            return


        if (
            left_area > MAX_BOARD_AREA_RATIO
            or
            right_area > MAX_BOARD_AREA_RATIO
        ):

            self.reset_stability()


            if (
                now
                -
                self.last_status_time
                >
                1.5
            ):

                self.get_logger().info(
                    "Plansza za duza w obrazie. "
                    "Odsun ja."
                )

                self.last_status_time = now


            return


        # ====================================================
        # DESCRIPTOR
        # ====================================================

        left_descriptor = self.make_descriptor(
            left_corners,
            width,
            height,
        )

        right_descriptor = self.make_descriptor(
            right_corners,
            width,
            height,
        )


        descriptor = (
            left_descriptor
            +
            right_descriptor
        ) / 2.0


        # ====================================================
        # STABILNOSC
        # ====================================================

        stable = self.update_stability(
            descriptor
        )


        if not stable:

            if (
                now
                -
                self.last_status_time
                >
                1.0
            ):

                self.get_logger().info(
                    f"Plansza wykryta. "
                    f"TRZYMAJ NIERUCHOMO... "
                    f"{self.stable_counter}/"
                    f"{STABLE_FRAMES_REQUIRED}"
                )

                self.last_status_time = now


            return


        # ====================================================
        # ROZNORODNOSC
        # ====================================================

        if not self.is_diverse(
            descriptor
        ):

            self.reset_stability()


            if (
                now
                -
                self.last_status_time
                >
                1.0
            ):

                self.get_logger().info(
                    "Pozycja za podobna. "
                    "PRZESUN / PRZECHYL / ODDAL plansze."
                )

                self.last_status_time = now


            return


        # ====================================================
        # AKCEPTACJA
        # ====================================================

        self.object_points.append(
            self.object_template.copy()
        )

        self.left_points.append(
            left_corners.copy()
        )

        self.right_points.append(
            right_corners.copy()
        )

        self.accepted_descriptors.append(
            descriptor.copy()
        )


        sample_number = len(
            self.object_points
        )


        self.save_sample_images(
            left,
            right,
            left_corners,
            right_corners,
            sample_number,
        )


        self.get_logger().info(
            ""
        )

        self.get_logger().info(
            "================================================"
        )

        self.get_logger().info(
            f"PARA ZAAKCEPTOWANA: "
            f"{sample_number}/{TARGET_SAMPLES}"
        )

        self.get_logger().info(

        if sample_number >= TARGET_SAMPLES:

            self.perform_calibration()

    #
    # V3:
    # - fx = fy
    # - principal point w centrum
    # - p1 = p2 = 0
    # - k3 = 0

    def calibrate_single_camera(
        self,
        image_points,
    ):

        width, height = (
            self.image_size
        )


        camera_matrix = (
            cv2.initCameraMatrix2D(
                self.object_points,
                image_points,
                self.image_size,
                aspectRatio=1.0,
            )
        )


        # Principal point:
        camera_matrix[0, 2] = (
            width - 1
        ) / 2.0

        camera_matrix[1, 2] = (
            height - 1
        ) / 2.0


        # Kwadratowe piksele:
        focal = float(
            (
                camera_matrix[0, 0]
                +
                camera_matrix[1, 1]
            )
            /
            2.0
        )

        camera_matrix[0, 0] = focal
        camera_matrix[1, 1] = focal


        distortion = np.zeros(
            (
                5,
                1
            ),
            dtype=np.float64,
        )


        flags = (
            cv2.CALIB_USE_INTRINSIC_GUESS
            |
            cv2.CALIB_FIX_ASPECT_RATIO
            |
            cv2.CALIB_FIX_PRINCIPAL_POINT
            |
            cv2.CALIB_ZERO_TANGENT_DIST
            |
            cv2.CALIB_FIX_K3
        )


        criteria = (
            cv2.TERM_CRITERIA_EPS
            |
            cv2.TERM_CRITERIA_MAX_ITER,
            200,
            1e-9,
        )


        return cv2.calibrateCamera(
            self.object_points,
            image_points,
            self.image_size,
            camera_matrix,
            distortion,
            flags=flags,
            criteria=criteria,
        )

    # ========================================================
    # AI CAMERA POINT
    # ========================================================

    def camera_callback(
        self,
        msg,
    ):

        point = np.array(
            [
                msg.point.x,
                msg.point.y,
                msg.point.z,
            ],
            dtype=np.float64,
        )

        if not np.all(
            np.isfinite(
                point
            )
        ):

            return

        if (
            point[2] < 0.10
            or
            point[2] > 2.00
        ):

            return

        self.camera_buffer.append(
            point
        )


    # ========================================================
    # TOOL POSITION
    # ========================================================

    def get_tool_position(
        self,
    ):

        try:

            transform = (
                self.tf_buffer
                .lookup_transform(
                    BASE_FRAME,
                    TOOL_FRAME,
                    rclpy.time.Time(),
                )
            )

        except Exception as exc:

            self.get_logger().error(
                f"Brak TF "
                f"{BASE_FRAME}->{TOOL_FRAME}: "
                f"{exc}"
            )

            return None

        t = (
            transform
            .transform
            .translation
        )

        return np.array(
            [
                t.x,
                t.y,
                t.z,
            ],
            dtype=np.float64,
        )


    # ========================================================
    # STABLE AI POINT
    # ========================================================

    def get_stable_camera_point(
        self,
    ):

        if (
            len(
                self.camera_buffer
            )
            <
            MIN_CAMERA_SAMPLES
        ):

            self.get_logger().error(
                "Za malo pomiarow AI. "
                "Poczekaj 2-3 sekundy."
            )

            return None

        points = np.array(
            self.camera_buffer,
            dtype=np.float64,
        )

        median = np.median(
            points,
            axis=0,
        )

        distances = np.linalg.norm(
            points
            -
            median,
            axis=1,
        )

        threshold = np.percentile(
            distances,
            70,
        )

        filtered = points[
            distances
            <=
            threshold
        ]

        if (
            filtered.shape[0]
            <
            6
        ):

            self.get_logger().error(
                "Za malo stabilnych pomiarow AI."
            )

            return None

        final_point = np.median(
            filtered,
            axis=0,
        )

        minimum = np.min(
            filtered,
            axis=0,
        )

        maximum = np.max(
            filtered,
            axis=0,
        )

        spread = (
            maximum
            -
            minimum
        )

        self.get_logger().info(
            "Stabilnosc AI:"
        )

        self.get_logger().info(
            f"X spread = "
            f"{spread[0] * 1000.0:.1f} mm"
        )

        self.get_logger().info(
            f"Y spread = "
            f"{spread[1] * 1000.0:.1f} mm"
        )

        self.get_logger().info(
            f"Z spread = "
            f"{spread[2] * 1000.0:.1f} mm"
        )

        if (
            np.max(
                spread
            )
            >
            MAX_CAMERA_SPREAD_M
        ):

            self.get_logger().error(
                "AI jest zbyt niestabilne. "
                "Nie zapisuje punktu."
            )

            return None

        return final_point

    # ========================================================
    # ROTATION -> RPY
    # ========================================================

    @staticmethod
    def rotation_to_rpy(
        rotation,
    ):

        sy = math.sqrt(
            rotation[0, 0] ** 2
            +
            rotation[1, 0] ** 2
        )

        singular = (
            sy < 1e-6
        )

        if not singular:

            roll = math.atan2(
                rotation[2, 1],
                rotation[2, 2],
            )

            pitch = math.atan2(
                -rotation[2, 0],
                sy,
            )

            yaw = math.atan2(
                rotation[1, 0],
                rotation[0, 0],
            )

        else:

            roll = math.atan2(
                -rotation[1, 2],
                rotation[1, 1],
            )

            pitch = math.atan2(
                -rotation[2, 0],
                sy,
            )

            yaw = 0.0

        return (
            roll,
            pitch,
            yaw,
        )


    # ========================================================
    # SOLVE
    # ========================================================

    def solve(
        self,
    ):

        count = len(
            self.camera_samples
        )

        if count < 5:

            self.get_logger().error(
                "Potrzeba minimum 5 gotowych probek. "
                "Najlepiej 6-8."
            )

            return

        camera = np.array(
            self.camera_samples,
            dtype=np.float64,
        )

        base = np.array(
            self.base_samples,
            dtype=np.float64,
        )

        camera_center = np.mean(
            camera,
            axis=0,
        )

        base_center = np.mean(
            base,
            axis=0,
        )

        camera_zero = (
            camera
            -
            camera_center
        )

        base_zero = (
            base
            -
            base_center
        )

        covariance = (
            camera_zero.T
            @
            base_zero
        )

        u, _, vt = np.linalg.svd(
            covariance
        )

        rotation = (
            vt.T
            @
            u.T
        )

        if (
            np.linalg.det(
                rotation
            )
            <
            0.0
        ):

            vt[
                -1,
                :
            ] *= -1.0

            rotation = (
                vt.T
                @
                u.T
            )

        translation = (
            base_center
            -
            rotation
            @
            camera_center
        )

        predicted = (
            (
                rotation
                @
                camera.T
            ).T
            +
            translation
        )

        errors = np.linalg.norm(
            predicted
            -
            base,
            axis=1,
        )

        rms = math.sqrt(
            float(
                np.mean(
                    errors ** 2
                )
            )
        )

        maximum = float(
            np.max(
                errors
            )
        )

        roll, pitch, yaw = (
            self.rotation_to_rpy(
                rotation
            )
        )

        self.get_logger().info(
            ""
        )

        self.get_logger().info(
            "=========================================="
        )

        self.get_logger().info(
            "WYNIK KALIBRACJI AI"
        )

        self.get_logger().info(
            "=========================================="
        )

        for index, error in enumerate(
            errors
        ):

            self.get_logger().info(
                f"Probka {index + 1}: "
                f"{error * 1000.0:.2f} mm"
            )

        self.get_logger().info(
            ""
        )

        self.get_logger().info(
            "TRANSLATION [m]:"
        )

        self.get_logger().info(
            f"x = {translation[0]:+.8f}"
        )

        self.get_logger().info(
            f"y = {translation[1]:+.8f}"
        )

        self.get_logger().info(
            f"z = {translation[2]:+.8f}"
        )

        self.get_logger().info(
            ""
        )

        self.get_logger().info(
            "RPY [rad]:"
        )

        self.get_logger().info(
            f"roll  = {roll:+.8f}"
        )

        self.get_logger().info(
            f"pitch = {pitch:+.8f}"
        )

        self.get_logger().info(
            f"yaw   = {yaw:+.8f}"
        )

        self.get_logger().info(
            ""
        )

        self.get_logger().info(
            "RPY [deg]:"
        )

        self.get_logger().info(
            f"roll  = "
            f"{math.degrees(roll):+.3f}"
        )

        self.get_logger().info(
            f"pitch = "
            f"{math.degrees(pitch):+.3f}"
        )

        self.get_logger().info(

        )

        self.get_logger().info(

        self.get_logger().info(
